function formatCooldown(nextAt, label){

function normSearch(s){return String(s||'').toLowerCase().trim()}

function buildAlbumIndex(){
  const byKey=new Map();
  state.tracks.forEach(t=>{
    const album=(t.album||'').trim();
    if(!album)return;
    const key=(t.artistId||t.artist||'')+'::'+album.toLowerCase();
    if(!byKey.has(key)){
      byKey.set(key,{id:t.albumId||0,name:album,artist:t.artist||'',artistId:t.artistId||0,coverUrl:t.coverUrl||'',genre:t.genre||'',tracks:0,plays:0});
    }
    const a=byKey.get(key);
    a.tracks+=1;
    a.plays+=Number(t.playCount)||0;
    if(!a.coverUrl&&t.coverUrl)a.coverUrl=t.coverUrl;
  });
  return [...byKey.values()].sort((a,b)=>(b.plays-a.plays)||a.name.localeCompare(b.name,'ru'));
}

function findArtistForName(name){
  const n=String(name||'').trim().toLowerCase();

  return state.artists.find(a=>String(a.name||'').trim().toLowerCase()===n) || null;
}
function getClickableTrackArtists(t){
  if(Array.isArray(t.artists) && t.artists.length){
    return t.artists.map(a=>({id:Number(a.id)||0,name:a.name||''})).filter(a=>a.name);
  }
  const names=String(t.artist||'').split(',').map(v=>v.trim()).filter(Boolean);
  if(!names.length && t.artist) names.push(String(t.artist));
  return names.map((name,idx)=>{
    const match=findArtistForName(name);
    const isMain=idx===0 && t.artistId;
    return {id:match?.id || (isMain ? Number(t.artistId)||0 : 0), name};
  });
}

}

function closeTrackMenus(){
  document.querySelectorAll('.track-menu.open').forEach(m=>m.classList.remove('open'));
  document.querySelectorAll('.track-actions.open').forEach(a=>a.classList.remove('open'));

function toggleTrackMenu(event,trackId,targetId){
  event.stopPropagation();
  const menu=qs(`trackMenu-${targetId}-${trackId}`);
  const actions=menu?.closest('.track-actions');
  const btn=event.currentTarget;
  const wasOpen=menu?.classList.contains('open');
  closeTrackMenus();
  if(menu && !wasOpen){
    menu.classList.add('open');
    actions?.classList.add('open');
    const r=btn.getBoundingClientRect();
    const mw=Math.min(342, window.innerWidth-24);
    let left=Math.min(window.innerWidth-mw-12, Math.max(12, r.right-mw));
    let top=r.bottom+8;
    menu.style.width=mw+'px';
    menu.style.left=left+'px';
    menu.style.top=top+'px';
    requestAnimationFrame(()=>{
      const mr=menu.getBoundingClientRect();
      if(mr.bottom>window.innerHeight-12){
        menu.style.top=Math.max(12, r.top-mr.height-8)+'px';
      }
    });
  }
}

function playTrackNext(trackId){
  const t=findTrackById(trackId);
  if(!t)return toast('Трек не найден','err');
  state.queue=state.queue.filter(x=>Number(x.id)!==Number(t.id));
  state.queue.unshift(t);
  toast('Трек будет играть следующим','ok');
}
function addTrackToQueue(trackId){
  const t=findTrackById(trackId);
  if(!t)return toast('Трек не найден','err');
  if(state.queue.some(x=>Number(x.id)===Number(t.id))) return toast('Этот трек уже в очереди','err');
  state.queue.push(t);
  try{localStorage.setItem('venyl_queue_v2',JSON.stringify(state.queue.slice(0,100)))}catch{}
  toast('Добавлено в очередь','ok');
}
function excludeTrack(trackId){
  state.excludedTracks.add(Number(trackId));
  localStorage.setItem('venyl_excluded_tracks',JSON.stringify([...state.excludedTracks]));
  toast('Трек исключён из музыкальных предпочтений','ok');
}
function startTrackRadio(trackId){
  const base=findTrackById(trackId);
  if(!base)return toast('Трек не найден','err');
  const genre=String(base.genre||'').toLowerCase();
  const artist=String(base.artist||'').toLowerCase();
  const radio=state.tracks.filter(t=>Number(t.id)!==Number(trackId) && (String(t.artist||'').toLowerCase()===artist || (genre && String(t.genre||'').toLowerCase().includes(genre))));
  const list=[base,...radio.sort(()=>Math.random()-0.5)];
  playTrackObject(list[0],0,list);
  toast('Запущено радио по треку','ok');
}

async function shareTrack(trackId){
  const t=findTrackById(trackId);
  if(!t)return toast('Трек не найден','err');
  const text=`${t.title||'Трек'} — ${t.artist||'Venyl'}`;
  const url=window.location.href;
  try{
    if(navigator.share){await navigator.share({title:text,text,url});}
    else if(navigator.clipboard){await navigator.clipboard.writeText(`${text} ${url}`);toast('Ссылка скопирована','ok');}
    else{toast('Ссылка: '+url);}
  }catch{toast('Не удалось поделиться','err')}
}

function playTrackObject(t,idx,list){
  if(!t)return;
  state.currentIdx=idx;state.currentTrack=t;state.playList=list||state.filtered;
  updatePlayerMeta();
  if(t.audioUrl){
    const shouldReload=audio.src!==new URL(t.audioUrl, window.location.origin).href;
    if(shouldReload) audio.src=t.audioUrl;
    audio.play().then(()=>{state.isPlaying=true;updatePlayBtn();qs('eqBars').style.display='flex';registerListen(t.id);renderTracks(state.filtered,'tracksList');if(state.currentArtist)renderTracks(state.currentArtist.tracks,'artistTracksList');if(state.currentAlbum)renderTracks(state.currentAlbum.tracks,'albumTracksList');renderTracks(state.subscribedTracks,'subscribedTracksList');renderTracks(state.favoriteTracks,'favoriteTracksList');if(state.currentPlaylist)renderTracks(state.currentPlaylist.tracks,'playlistTracksList');updatePlayerMeta();})
    .catch(()=>{state.isPlaying=false;updatePlayBtn();toast('Не удалось воспроизвести трек','err')});
  }
}
function updatePlayerMeta(){
  const t=state.currentTrack;
  qs('npCover').innerHTML=t?(t.coverUrl?`<img src="${t.coverUrl}" style="width:100%;height:100%;object-fit:cover">`:'<svg viewBox="0 0 20 20" fill="currentColor"><path d="M18 3a1 1 0 00-1.196-.98l-10 2A1 1 0 006 5v9.114A4.369 4.369 0 005 14c-1.657 0-3 .895-3 2s1.343 2 3 2 3-.895 3-2V7.82l8-1.6v5.894A4.37 4.37 0 0015 12c-1.657 0-3 .895-3 2s1.343 2 3 2 3-.895 3-2V3z"/></svg>'):'';
  qs('npTitle').textContent=t?t.title:'Выберите трек';
  qs('npArtist').textContent=t?(t.artist||'—'):'—';
  qs('favBtn').classList.toggle('active', !!(t && state.favorites.has(Number(t.id))));
}

function nextTrack(){
  if(state.queue.length){
    const next=state.queue.shift();
    const l=state.playList||state.filtered;
    const idx=l.findIndex(t=>Number(t.id)===Number(next.id));
    playTrackObject(next,idx>=0?idx:0,l.length?l:[next]);
    return;
  }
  const l=state.playList||state.filtered;if(!l.length)return;const i=(state.currentIdx+1)%l.length;playTrackObject(l[i],i,l)
}
